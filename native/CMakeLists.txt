cmake_minimum_required(VERSION 3.24)
project(hdr_abc LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(IMATH_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(IMATH_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(PYTHON OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/imath EXCLUDE_FROM_ALL)
set(ALEMBIC_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(USE_TESTS OFF CACHE BOOL "" FORCE)
set(USE_BINARIES OFF CACHE BOOL "" FORCE)
set(USE_HDF5 OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/alembic EXCLUDE_FROM_ALL)
add_library(hdr_abc SHARED bridge.cpp)
target_link_libraries(hdr_abc PRIVATE Alembic)
target_compile_definitions(hdr_abc PRIVATE HDR_ABC_BUILD)
add_executable(abc_fixture fixture.cpp)
target_link_libraries(abc_fixture PRIVATE Alembic)
